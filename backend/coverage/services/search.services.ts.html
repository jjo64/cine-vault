export function levenshtein(a: string, b: string): number {
  const source = a.toLowerCase()
  const target = b.toLowerCase()
  const m = source.length
  const n = target.length
 
  const dp = Array.from({ length: m + 1 }, (_, i) =>
    Array.from({ length: n + 1 }, (_, j) => (i === 0 ? j : j === 0 ? i : 0))
  )
 
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (source[i - 1] === target[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1]
      } else {
        dp[i][j] = 1 + Math.min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
      }
    }
  }
 
  return dp[m][n]
}

export async function enriquecerConDatosLocales(tmdbIds: number[]): Promise<Map<number, LocalCounts>> {

    _count: {
      vault: number
      reviews: number
      watchlist: number
    }
  }> = []
 
  try {
    refs = await prisma.movies_ref.findMany({
      where: { tmdb_id: { in: tmdbIds } },
      select: {
        tmdb_id: true,
        _count: {
          select: {
            vault: true,
            reviews: true,
            watchlist: true,
          },
        },
      },
    })
  } catch (error) {
    const maybePrismaError = error as { code?: string }
    if (maybePrismaError?.code === "P2022") {
      return new Map()
    }
    throw error
  }
 
  const mapa = new Map<number, LocalCounts>()

    mapa.set(ref.tmdb_id, {
      vault_count: ref._count.vault,
      review_count: ref._count.reviews,
      watchlist_count: ref._count.watchlist,
    })
  }
 
  return mapa
}
